// Call after receiving an OAuth code through the Pawdy callback page.
// The code is single-use. Do not paste it into the Sheet.
function exchangeTikTokOrganicCode(authorizationCode) {
  requireValidatedIntegration_();
  requireText_(authorizationCode, 'authorizationCode');
  var p = props_();
  var payload = {
    client_key: requireProp_('TIKTOK_ORGANIC_CLIENT_KEY'),
    client_secret: requireProp_('TIKTOK_ORGANIC_CLIENT_SECRET'),
    code: authorizationCode,
    grant_type: 'authorization_code',
    redirect_uri: requireProp_('TIKTOK_ORGANIC_REDIRECT_URI')
  };
  var data = formPost_(TT.ORGANIC_TOKEN_URL, payload);
  saveToken_('TIKTOK_ORGANIC', data);
  return safeTokenResult_(data);
}

function refreshTikTokOrganicToken_() {
  var data = formPost_(TT.ORGANIC_TOKEN_URL, {
    client_key: requireProp_('TIKTOK_ORGANIC_CLIENT_KEY'),
    client_secret: requireProp_('TIKTOK_ORGANIC_CLIENT_SECRET'),
    refresh_token: requireProp_('TIKTOK_ORGANIC_REFRESH_TOKEN'),
    grant_type: 'refresh_token'
  });
  saveToken_('TIKTOK_ORGANIC', data);
  return data.access_token;
}

// Marketing API uses a distinct Business API app/approval from organic data.
function exchangeTikTokAdsCode(authorizationCode) {
  requireValidatedIntegration_();
  requireText_(authorizationCode, 'authorizationCode');
  var data = jsonFetch_(TT.ADS_TOKEN_URL, {
    app_id: requireProp_('TIKTOK_ADS_APP_ID'),
    secret: requireProp_('TIKTOK_ADS_APP_SECRET'),
    auth_code: authorizationCode
  });
  saveToken_('TIKTOK_ADS', data);
  return safeTokenResult_(data);
}

function token_(prefix) {
  var p = props_();
  var expires = Number(p.getProperty(prefix + '_TOKEN_EXPIRES_AT') || 0);
  if (prefix === 'TIKTOK_ADS') {
    if (expires && Date.now() >= expires) throw new Error('Business token expired; reauthorization required');
    return requireProp_('TIKTOK_ADS_ACCESS_TOKEN');
  }
  if (!p.getProperty(prefix + '_ACCESS_TOKEN') || Date.now() > expires - 5 * 60 * 1000) {
    return refreshTikTokOrganicToken_();
  }
  return p.getProperty(prefix + '_ACCESS_TOKEN');
}

/* ---------- Main sync ---------- */

function syncTikTokNow() {
  requireValidatedIntegration_();
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(5000)) return;
  try {
    var sheet = getTikTokSheet_();
    var existing = readExisting_(sheet);
    var result = { organic: 'ไม่ได้ตั้งค่า', ads: 'ไม่ได้ตั้งค่า', errors: [] };

    if (hasProps_(['TIKTOK_ORGANIC_CLIENT_KEY', 'TIKTOK_ORGANIC_CLIENT_SECRET', 'TIKTOK_ORGANIC_REFRESH_TOKEN'])) {
      try {
        mergeOrganic_(existing, fetchOrganicVideos_());
        result.organic = 'API อัปเดตแล้ว';
      } catch (error) { throw new Error('Organic fetch failed; no Sheet write performed'); }
    }

    if (hasProps_(['TIKTOK_ADS_ACCESS_TOKEN', 'TIKTOK_ADVERTISER_ID'])) {
      try {
        mergeAds_(existing, fetchAdsReport_());
        result.ads = 'API อัปเดตแล้ว';
      } catch (error) { throw new Error('Ads fetch failed; no Sheet write performed'); }
    }

    if (result.organic === 'ไม่ได้ตั้งค่า' && result.ads === 'ไม่ได้ตั้งค่า') throw new Error('No API configured; no Sheet write performed');
    writeRows_(sheet, existing, result);
    updateDashboardNote_(result);
    if (result.errors.length) throw new Error(result.errors.join(' | '));
    return result;
  } finally { lock.releaseLock(); }
}

function testTikTokConnection() {
  var report = syncTikTokNow();
  toast_('TikTok: Organic ' + report.organic + ' / Ads ' + report.ads);
}

/* ---------- TikTok API reads ---------- */

function fetchOrganicVideos_() {
  if (requireProp_('TIKTOK_ORGANIC_PRODUCT') !== 'DISPLAY') throw new Error('Accounts API adapter not configured');
  var fields = [
    'id', 'create_time', 'title', 'video_description', 'cover_image_url', 'share_url',
    'view_count', 'like_count', 'comment_count', 'share_count'
  ].join(',');
  var all = [], cursor, more = true, seen = {}, pages = 0;
  while (more) {
    if (++pages > 100) throw new Error('Pagination limit; refusing partial data');
    var body = { max_count: 20 };
    if (cursor !== undefined) body.cursor = cursor;
    var json = jsonFetch_(TT.ORGANIC_VIDEO_URL + '?fields=' + encodeURIComponent(fields), body, {
      Authorization: 'Bearer ' + token_('TIKTOK_ORGANIC')
    });
    var data = json || {};
    if (!Array.isArray(data.videos) || typeof data.has_more !== 'boolean') throw new Error('Invalid video response');
    all = all.concat(data.videos);
    more = data.has_more;
    if (more) {
      cursor = data.cursor;
      if (!Number.isSafeInteger(cursor) || seen[cursor]) throw new Error('Invalid pagination cursor');
      seen[cursor] = true;
    }
  }
  var videos = all.map(normalizeOrganic_).filter(function (v) { return v.date.getTime() >= PAWDY.since*1000 && v.date.getTime() < PAWDY.until*1000; });
  if (new Set(videos.map(function (v) { return v.postId; })).size !== videos.length) throw new Error('Duplicate API Post ID');
  if (!videos.length) throw new Error('No videos in configured period; preserve existing data');
  return videos;
}

function fetchAdsReport_() {
  var start = requireProp_('TIKTOK_REPORT_START');
  var end = requireProp_('TIKTOK_REPORT_END');
  if (!/^\d{4}-\d{2}-\d{2}$/.test(start) || !/^\d{4}-\d{2}-\d{2}$/.test(end) || start > end) throw new Error('Invalid report period');
  var payload = {
    advertiser_id: requireProp_('TIKTOK_ADVERTISER_ID'),
    service_type: 'AUCTION', report_type: 'BASIC', data_level: 'AUCTION_AD',
    dimensions: JSON.stringify(['ad_id']),
    metrics: JSON.stringify(['spend', 'impressions', 'clicks', 'video_play_actions', 'video_watched_2s', 'video_watched_6s', 'video_views_p100']),
    start_date: start,
    end_date: end, page: 1, page_size: 1000
  };
  var list = [], pages = 1;
  do {
    var json = getJson_(TT.ADS_REPORT_URL, payload, { 'Access-Token': token_('TIKTOK_ADS') });
    if (!Array.isArray(json.list) || !json.page_info) throw new Error('Invalid report response');
    pages = Number(json.page_info.total_page);
    if (!Number.isInteger(pages) || pages < 1 || pages > 100) throw new Error('Invalid report pagination');
    list = list.concat(json.list);
    payload.page++;
  } while (payload.page <= pages);
  // Ads metrics cannot safely be attributed to an organic post without a video/post ID.
  // Set a mapping in Script Properties: TIKTOK_AD_TO_POST_JSON = {"ad_id":"post_id"}.
  var mapping = parseJsonProp_('TIKTOK_AD_TO_POST_JSON', {});
  return list.map(function (row) { return normalizeAd_(row, mapping); });
}

function normalizeOrganic_(video) {
  if (!Number.isSafeInteger(video.create_time) || video.create_time <= 0) throw new Error('Invalid video timestamp');
  return {
    postId: exactId_(video.id), date: toDate_(video.create_time), link: video.share_url || '',
    preview: video.cover_image_url || '', caption: video.video_description || video.title || '',
    views: num_(video.view_count), likes: num_(video.like_count), comments: num_(video.comment_count),
    shares: num_(video.share_count), updated: new Date(), source: 'TikTok API'
  };
}

function normalizeAd_(row, mapping) {
  var dim = row.dimensions || {}, met = row.metrics || {};
  var adId = exactId_(dim.ad_id || row.ad_id);
  return {
    postId: mapping[adId] ? exactId_(mapping[adId]) : '', adId: adId,
    impressions: num_(met.impressions), views: num_(met.video_play_actions), starts: num_(met.video_play_actions),
    views2s: num_(met.video_watched_2s), views6s: num_(met.video_watched_6s),
    completions: num_(met.video_views_p100), clicks: num_(met.clicks), spend: num_(met.spend)
  };
}

/* ---------- Sheet operations ---------- */

function getTikTokSheet_() {
  var ss = SpreadsheetApp.openById(PAWDY.spreadsheetId);
  if (ss.getSpreadsheetTimeZone() !== PAWDY.timezone) throw new Error('Unexpected spreadsheet timezone');
  var sheet = ss.getSheetByName(TT.SHEET);
  if (!sheet) throw new Error('Target sheet missing');
  return sheet;
}

function readExisting_(sheet) {
  var all = sheet.getDataRange().getValues();
  var header = headerRow_(all), values = all.slice(header + 1);
  var byId = Object.create(null);
  values.forEach(function (row) {
    if (!row[1]) { if (row.some(function (x) { return x !== ''; })) throw new Error('Data row missing ID'); return; }
    var id = exactId_(row[1]);
    if (byId[id]) throw new Error('Duplicate Post ID');
    byId[id] = row.slice(0, TT_COLUMNS.length);
  });
  Object.defineProperty(byId, '_layout', { value: {
    header: header, positions: Object.keys(byId).reduce(function (out, id) {
      out[id] = all.findIndex(function (row, index) { return index > header && row[1] === id; });
      return out;
    }, Object.create(null)), end: header + 1 + values.length
  }});
  return byId;
}

function headerRow_(rows) {
  var matches = [];
  rows.slice(0, 20).forEach(function (row, index) {
    if (TT_COLUMNS.every(function (name, col) { return row[col] === name; })) matches.push(index);
  });
  if (matches.length !== 1) throw new Error('Ambiguous or incompatible table header');
  return matches[0];
}

function blankRow_() { return TT_COLUMNS.map(function () { return ''; }); }

function mergeOrganic_(rows, videos) {
  videos.forEach(function (v) {
    if (!v.postId) return;
    var row = rows[v.postId] || (rows[v.postId] = blankRow_());
    row[0] = v.date || row[0]; row[1] = v.postId; row[2] = v.link; row[3] = v.preview;
    row[4] = v.caption; row[5] = v.views; row[6] = v.likes; row[7] = v.comments; row[8] = v.shares;
    row[10] = v.updated; row[11] = v.source;
    row.ttOrganic = true;
  });
}

function mergeAds_(rows, ads) {
  var grouped = Object.create(null), seen = Object.create(null);
  ads.forEach(function (a) {
    if (!a.postId || !rows[a.postId]) throw new Error('Unmapped ad; report not applied');
    if (seen[a.adId]) throw new Error('Duplicate ad in report');
    seen[a.adId] = true;
    if (!grouped[a.postId]) { grouped[a.postId] = Object.assign({}, a); return; }
    ['spend','impressions','views','starts','views2s','views6s','completions','clicks'].forEach(function (key) {
      grouped[a.postId][key] = sumKnown_(grouped[a.postId][key], a[key]);
    });
  });
  Object.keys(grouped).forEach(function (id) {
    var a = grouped[id], row = rows[id];
    row[1] = a.postId; row[9] = a.spend; row[12] = a.impressions; row[13] = a.views;
    row[14] = a.starts; row[15] = a.views2s; row[16] = a.views6s; row[17] = a.completions;
    row[18] = a.clicks; row[28] = a.spend; row[35] = 'TikTok Ads API';
    row.ttAds = true;
  });
}

function writeRows_(sheet, byId, result) {
  requireValidatedIntegration_();
  var requests = tikTokRequests_(sheet, byId);
  if (!requests.length) throw new Error('No new API data');
  var response;
  try {
    response = UrlFetchApp.fetch('https://sheets.googleapis.com/v4/spreadsheets/' + PAWDY.spreadsheetId + ':batchUpdate', {
      method: 'post', contentType: 'application/json', followRedirects: false,
      headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
      payload: JSON.stringify({requests: requests}), muteHttpExceptions: true
    });
  } catch (_) { throw new Error('Sheet write outcome unknown; inspect before retry'); }
  if (response.getResponseCode() !== 200) throw new Error('Sheet batch rejected');
}

// Build a single atomic batch. Only API-owned cells are replaced. Sort includes
// all columns so manually maintained business data stays attached to its ID.
function tikTokRequests_(sheet, rows) {
  var layout = rows._layout;
  if (!layout) throw new Error('Read table layout before planning writes');
  var sheetId = sheet.getSheetId(), requests = [], end = layout.end;
  var ids = Object.keys(rows), added = ids.filter(function (id) { return layout.positions[id] === undefined; });
  var width = Math.max(sheet.getLastColumn(), TT_COLUMNS.length);
  if (added.length) {
    requests.push({insertDimension:{range:{sheetId:sheetId,dimension:'ROWS',startIndex:end,endIndex:end+added.length},inheritFromBefore:true}});
    added.forEach(function (id, offset) { layout.positions[id] = end + offset; });
    end += added.length;
  }
  ids.forEach(function (id) {
    var row = rows[id], at = layout.positions[id];
    if (row.ttOrganic) {
      requests.push(ttCells_(sheetId, at, 0, row.slice(0, 9)));
      requests.push(ttCells_(sheetId, at, 10, row.slice(10, 12)));
    }
    if (row.ttAds) {
      requests.push(ttCells_(sheetId, at, 9, [row[9]]));
      requests.push(ttCells_(sheetId, at, 12, row.slice(12, 19)));
      requests.push(ttCells_(sheetId, at, 35, [row[35]]));
    }
    if (row.ttOrganic || row.ttAds) {
      // KPI values refreshed together with their input data; raw Conversions and
      // Revenue are retained. No paid CPE is inferred from organic engagement.
      requests.push(ttCells_(sheetId, at, 21, calculateRow_(row).slice(21, 35)));
    }
  });
  if (requests.length && end > layout.header + 1) requests.push({sortRange:{
    range:{sheetId:sheetId,startRowIndex:layout.header+1,endRowIndex:end,startColumnIndex:0,endColumnIndex:width},
    sortSpecs:[{dimensionIndex:0,sortOrder:'DESCENDING'}]
  }});
  return requests;
}

function ttCells_(sheetId, rowIndex, columnIndex, values) {
  return {updateCells:{start:{sheetId:sheetId,rowIndex:rowIndex,columnIndex:columnIndex},rows:[{
    values: values.map(function (value) {
      if (value instanceof Date) {
        // Inherit the configured report timezone, not the machine's timezone.
        value = Utilities.formatDate(value, PAWDY.timezone, 'yyyy-MM-dd HH:mm:ss');
      }
      return {userEnteredValue: typeof value === 'number' ? {numberValue:value} : {stringValue:String(value === undefined || value === null ? '' : value)}};
    })
  }],fields:'userEnteredValue'}};
}

function calculateRow_(r) {
  var v = num_(r[5]), l = num_(r[6]), c = num_(r[7]), s = num_(r[8]);
  var spend = num_(r[28]) === '' ? num_(r[9]) : num_(r[28]);
  var imp = num_(r[12]), pv = num_(r[13]), starts = num_(r[14]), v2 = num_(r[15]), v6 = num_(r[16]), complete = num_(r[17]), clicks = num_(r[18]);
  r[21] = ratio_(l, v); r[22] = ratio_(c, v); r[23] = ratio_(s, v); r[24] = ratio_(sumKnown_(sumKnown_(l, c), s), v);
  r[25] = ratio_(v2, starts); r[26] = ratio_(v6, starts); r[27] = ratio_(complete, starts);
  r[28] = spend; r[29] = ratio_(spend === '' ? '' : spend * 1000, imp); r[30] = ratio_(spend, pv);
  r[31] = ratio_(spend, v6); r[32] = ''; r[33] = ratio_(clicks, imp); r[34] = ratio_(num_(r[20]), spend);
  return r;
}

function updateDashboardNote_(result) {
  requireValidatedIntegration_();
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(TT.DASHBOARD);
  if (!sheet) return;
  // A non-sensitive audit trail, independent of any token or API response body.
  sheet.getRange('A1').setNote('TikTok ล่าสุด: ' + Utilities.formatDate(new Date(), TT.TZ, 'd MMM yyyy HH:mm') +
    ' | Organic: ' + result.organic + ' | Ads: ' + result.ads + (result.errors.length ? ' | ' + result.errors.join(' | ') : ''));
}

/* ---------- HTTP, properties and safe helpers ---------- */

function formPost_(url, payload, headers) {
  var response = UrlFetchApp.fetch(url, { method: 'post', contentType: 'application/x-www-form-urlencoded', payload: Object.keys(payload).map(function (k) { return encodeURIComponent(k)+'='+encodeURIComponent(payload[k]); }).join('&'), headers: headers || {}, muteHttpExceptions: true, followRedirects:false });
  return parseResponse_(response);
}

function getJson_(url, params, headers) {
  var query = Object.keys(params).map(function (key) { return encodeURIComponent(key) + '=' + encodeURIComponent(params[key]); }).join('&');
  return parseResponse_(UrlFetchApp.fetch(url + '?' + query, { method: 'get', headers: headers, muteHttpExceptions: true, followRedirects: false }));
}

function jsonFetch_(url, body, headers) {
  var response = UrlFetchApp.fetch(url, { method: 'post', contentType: 'application/json', payload: JSON.stringify(body), headers: headers || {}, muteHttpExceptions: true, followRedirects:false });
  return parseResponse_(response);
}

function parseResponse_(response) {
  var text = response.getContentText();
  var json;
  try { json = JSON.parse(text); } catch (_) { throw new Error('TikTok ส่งข้อมูลที่อ่านไม่ได้ (HTTP ' + response.getResponseCode() + ')'); }
  if (!json || typeof json !== 'object' || Array.isArray(json)) throw new Error('Invalid API envelope');
  var error = typeof json.error === 'string' ? Boolean(json.error) : json.error && json.error.code !== 'ok';
  if (json.code !== undefined && Number(json.code) !== 0) error = true;
  if (response.getResponseCode() < 200 || response.getResponseCode() >= 300 || error) throw new Error('TikTok request failed (HTTP ' + response.getResponseCode() + ')');
  return json.data || json;
}

function saveToken_(prefix, data) {
  var p = props_(), now = Date.now();
  if (!data.access_token) throw new Error('TikTok ไม่ส่ง access token กลับมา');
  p.setProperty(prefix + '_ACCESS_TOKEN', data.access_token);
  if (data.refresh_token) p.setProperty(prefix + '_REFRESH_TOKEN', data.refresh_token);
  if (Number(data.expires_in) > 0) p.setProperty(prefix + '_TOKEN_EXPIRES_AT', String(now + Number(data.expires_in) * 1000));
  else p.deleteProperty(prefix + '_TOKEN_EXPIRES_AT');
  if (data.refresh_expires_in) p.setProperty(prefix + '_REFRESH_EXPIRES_AT', String(now + Number(data.refresh_expires_in) * 1000));
}

function safeTokenResult_(data) { return { expires_in: data.expires_in || null, advertiser_ids: data.advertiser_ids || [] }; }
function props_() { return PropertiesService.getScriptProperties(); }
function requireProp_(name) { var v = props_().getProperty(name); if (!v) throw new Error('ยังไม่ได้ตั้ง Script Property: ' + name); return v; }
function hasProps_(names) { return names.every(function (name) { return Boolean(props_().getProperty(name)); }); }
function parseJsonProp_(name, fallback) { try { return JSON.parse(props_().getProperty(name) || JSON.stringify(fallback)); } catch (_) { throw new Error(name + ' ต้องเป็น JSON ที่ถูกต้อง'); } }
function requireText_(value, name) { if (!String(value || '').trim()) throw new Error(name + ' ว่างอยู่'); }
function num_(value) { if (value === null || value === undefined || typeof value === 'boolean' || String(value).trim() === '') return ''; var n = Number(value); return isFinite(n) && n >= 0 ? n : ''; }
function ratio_(a, b) { return a !== '' && b !== '' && b > 0 ? a / b : ''; }
function sumKnown_(a, b) { return a === '' || b === '' ? '' : a + b; }
function exactId_(value) { if (typeof value !== 'string' || !/^\d+$/.test(value)) throw new Error('ID must be decimal text'); return value; }
function toDate_(value) { var n = Number(value); return n ? new Date(n * (n < 1e11 ? 1000 : 1)) : ''; }
function cleanError_(error) { return String(error && error.message || error).replace(/(?:access|refresh)[_-]?token[^\s]*/ig, '[redacted]').slice(0, 400); }
function toast_(message) { try { SpreadsheetApp.getActiveSpreadsheet().toast(message, 'Pawdy TikTok'); } catch (_) { console.log(message); } }
